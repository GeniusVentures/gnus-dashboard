// lib/gnus/schema.ts
export type JobDefinition = Record<string, unknown>;
export const estimateSchema;      export type Estimate;
// { cost: baseUnits, gas: { gasPriceGwei: decimal, symbol: string (min 1) } | null, balance: baseUnits | null, network }
export const submitResultSchema;  export type SubmitResult;
// { jobId: string (min 1) | null, burnTxHash: TxHash, sdkAddress: SdkAddress, status: "submitted" | "bridgedNotProcessed" }
export const JOB_STATUSES: readonly ["submitted", "queued", "processing", "completed", "failed", "bridgedNotProcessed", "unknown"];
export type JobStatus = (typeof JOB_STATUSES)[number];
export const jobSchema;           export type Job;
// { jobId: string (min 1) | null, burnTxHash: TxHash, sdkAddress: SdkAddress, origin: string, idempotencyKey: string,
//   submittedAt: ISO 8601 with offset or Z, cost: baseUnits, status: JobStatus (unseen string -> "unknown") }
export const jobDetailSchema;     export type JobDetail;
// Job & { result: { cids: string[], data?: Record<string, unknown> } | null, error?: string }
export const jobListSchema;       export type JobList;   // { jobs: Job[], nextCursor: string | null }
export const earningsSchema;      export type Earnings;
// { sdkAddress: SdkAddress, balance: baseUnits, processing: { isProcessing: boolean, jobsProcessed: baseUnits | null },
//   earned: { total: baseUnits | null, history: Array<{ at: ISO 8601, amount: baseUnits, jobId: string }> | null } }
export const disconnectSchema;    // { code: integer, message: string }
export const eventSchemas: { gnus_accountsChanged: typeof accountsSchema; gnus_networkChanged: typeof networkSchema;
  gnus_jobUpdated: typeof jobSchema; disconnect: typeof disconnectSchema };
export type GnusEventName = keyof typeof eventSchemas;
export type GnusEventPayload<E extends GnusEventName>;   // z.infer of that event's schema

// lib/gnus/client.ts
export type GetJobParams =
  | { jobId: string; burnTxHash?: never; idempotencyKey?: never }
  | { burnTxHash: string; jobId?: never; idempotencyKey?: never }
  | { idempotencyKey: string; jobId?: never; burnTxHash?: never };
export interface GnusClient {
  getAccounts(): Promise<Accounts>;
  estimate(params: { job: JobDefinition }): Promise<Estimate>;
  submitJob(params: { job: JobDefinition; idempotencyKey: string }): Promise<SubmitResult>;
  listJobs(params?: { limit?: number; cursor?: string }): Promise<JobList>;
  getJob(params: GetJobParams): Promise<JobDetail | null>;
  getEarnings(params?: { includeChildren?: false }): Promise<Earnings>;
  /** Returns an unsubscribe function. */
  on<E extends GnusEventName>(event: E, handler: (payload: GnusEventPayload<E>) => void,
    onMismatch?: (err: GnusContractMismatchError) => void): () => void;
  close(): void;
}
